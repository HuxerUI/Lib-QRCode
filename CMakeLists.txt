cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME AND NOT CMAKE_TOOLCHAIN_FILE
        AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND "$ENV{MACOSX_DEPLOYMENT_TARGET}" STREQUAL "")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
endif()
project(lib_qrcode VERSION 0.1.0 LANGUAGES NONE)

option(HUXERUI_QRCODE_BUILD_VIEW "Build the HuxerUI QRCode and Barcode components" ON)
option(HUXERUI_QRCODE_BUILD_TESTS "Build barcode tests" OFF)
option(HUXERUI_QRCODE_BUILD_EXAMPLES "Build the barcode preview demo" OFF)
include("${CMAKE_CURRENT_SOURCE_DIR}/HuxerUIProject.cmake")
if(HUXERUI_PROJECT_PLAN_ONLY)
    return()
endif()

if(HUXERUI_QRCODE_BUILD_VIEW)
    huxerui_add_library(huxerui_qrcode SOURCES src/codec.cpp src/qrcode.cpp)
else()
    add_library(huxerui_qrcode STATIC src/codec.cpp)
endif()
add_library(HuxerUI::QRCode ALIAS huxerui_qrcode)
if(HUXERUI_LIBRARY_GRAPH_ONLY)
    return()
endif()

# Keep upstream configuration local, including static linkage and the dependency-free writer backend.
function(_huxerui_qrcode_add_zxing)
    set(BUILD_SHARED_LIBS OFF)
    set(ZXING_READERS ON)
    set(ZXING_WRITERS OLD)
    set(ZXING_C_API OFF)
    set(ZXING_EXPERIMENTAL_API OFF)
    set(ZXING_UNIT_TESTS OFF)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    add_subdirectory(third_party/zxing/core EXCLUDE_FROM_ALL)
endfunction()
if(TARGET ZXing OR TARGET ZXing::ZXing)
    message(FATAL_ERROR "Lib-QRCode uses its pinned ZXing backend; add Lib-QRCode before other ZXing consumers")
endif()
_huxerui_qrcode_add_zxing()

set_target_properties(huxerui_qrcode PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_compile_features(huxerui_qrcode PUBLIC cxx_std_20)
target_include_directories(huxerui_qrcode PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:include>")
target_link_libraries(huxerui_qrcode PRIVATE ZXing::ZXing)
target_compile_options(huxerui_qrcode PRIVATE
    "$<$<CXX_COMPILER_ID:MSVC>:/utf-8;/W4;/permissive->"
    "$<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Wpedantic>")
if(HUXERUI_QRCODE_BUILD_VIEW)
    target_link_libraries(huxerui_qrcode PUBLIC HuxerUI::huxerui)
endif()
if(HUXERUI_QRCODE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(HUXERUI_QRCODE_BUILD_EXAMPLES)
    if(NOT HUXERUI_QRCODE_BUILD_VIEW)
        message(FATAL_ERROR "The preview requires HUXERUI_QRCODE_BUILD_VIEW=ON")
    endif()
    add_subdirectory(examples/preview)
endif()
